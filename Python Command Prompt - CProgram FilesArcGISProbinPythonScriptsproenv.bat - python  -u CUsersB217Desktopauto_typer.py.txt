import time
import threading
import pyperclip
from pynput import keyboard
from pynput.keyboard import Controller, Key

typer = Controller()
busy = threading.Event()
stop = threading.Event()
CHAR_DELAY = 0.04


def type_clipboard():
    if busy.is_set():
        return

    text = pyperclip.paste()
    if not text:
        print("Clipboard is empty.")
        return

    busy.set()
    stop.clear()
    print(f"Typing {len(text)} characters...")

    try:
        for char in text:
            if stop.is_set():
                print("Typing stopped.")
                break

            # Handle line breaks explicitly
            if char == "\r":
                continue
            elif char == "\n":
                typer.press(Key.enter)
            elif char == "\t":
                typer.press(Key.tab)
            else:
                try:
                    typer.type(char)
                except Exception:
                    print(f"Skipped unsupported character: {char!r}")

            time.sleep(CHAR_DELAY)
        else:
            print("Finished typing.")

    finally:
        busy.clear()


def on_press(key):
    if key == Key.f8:
        if not busy.is_set():
            threading.Thread(
                target=type_clipboard,
                daemon=True
            ).start()

    elif key == Key.esc:
        stop.set()


print("Auto Typer is running.")
print("Copy text, focus the target field, then press F8.")
print("Press ESC to stop typing.")

with keyboard.Listener(on_press=on_press) as listener:
    listener.join()
